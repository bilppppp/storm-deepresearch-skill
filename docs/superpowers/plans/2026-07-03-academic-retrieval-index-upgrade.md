# Academic Retrieval Index Upgrade Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add receipt-bound academic search auditing, bibliographic verification, version-family deduplication, and corpus-seeded gap-fill to full dossiers while preserving the existing briefing path and STORM stage chain.

**Architecture:** Keep host-owned network retrieval and the current `plan -> retrieval -> findings` stages. Extend the existing retrieval input into a typed union, split it during `ingest` into one new audit artifact plus the existing capture manifest and source register, and enforce mode-specific gates from existing validators. No new Python module, CLI command, or user-facing mode is introduced.

**Tech Stack:** Python 3 standard library, existing `pypdf` dependency, JSON Schema draft 2020-12, `unittest`, existing Yao Meta Skill packaging and validation tools.

def validate_audit_snapshot(record: dict[str, Any], cache_root: Path) -> list[str]:
    try:
        snapshot = resolve_snapshot(cache_root, str(record.get("raw_artifact", "")))
    except SourceEvidenceError as exc:
        return [str(exc)]
    actual = sha256_bytes(snapshot.read_bytes())
    if actual != record.get("snapshot_sha256"):
        return ["snapshot_sha256 does not match audit artifact"]
    return []


def validate_candidate_snapshots(record: dict[str, Any], cache_root: Path) -> list[str]:
    errors: list[str] = []
    for outcome in record.get("resolver_outcomes", []):
        if not isinstance(outcome, dict) or outcome.get("status") == "skipped":
            continue
        errors.extend(validate_audit_snapshot({
            "raw_artifact": outcome.get("raw_artifact"),
            "snapshot_sha256": outcome.get("snapshot_sha256"),
        }, cache_root))
    return list(dict.fromkeys(errors))
```

def derive_bibliographic_status(candidate: dict[str, object]) -> str:
    outcomes = [item for item in candidate.get("resolver_outcomes", []) if isinstance(item, dict)]
    if any(item.get("status") == "matched" and item.get("metadata_match") is False for item in outcomes):
        return "conflicted"
    id_match = any(
        item.get("status") == "matched"
        and item.get("metadata_match") is True
        and item.get("query_basis") in {"doi", "pmid", "arxiv_id"}
        for item in outcomes
    )
    metadata_matches = {
        str(item.get("resolver"))
        for item in outcomes
        if item.get("status") == "matched" and item.get("metadata_match") is True
    }
    if id_match or len(metadata_matches) >= 2:
        return "verified"
    return "unverified"


def validate_version_families(candidates: list[dict[str, object]]) -> list[str]:
    families: dict[str, list[dict[str, object]]] = {}
    for candidate in candidates:
        family_id = candidate.get("version_family_id")
        if family_id:
            families.setdefault(str(family_id), []).append(candidate)
    errors: list[str] = []
    for family_id, members in families.items():
        if len(members) < 2:
            continue
        if any(item.get("relationship_basis") == "unresolved" for item in members):
            errors.append(f"{family_id} contains an unresolved version relationship")
    return errors
```
